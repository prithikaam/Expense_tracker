2. Average Monthly Expenditure Calculation

Demonstrates: AVG(), Nested Subquery with GROUP BY

SQL Aggregation 
SELECT 
    ROUND(AVG(monthly_total), 2) AS average_monthly_spending,
    COUNT(month_year) AS total_months_evaluated,
    SUM(monthly_total) AS total_overall_spent
FROM (
    SELECT 
        TO_CHAR(expense_date, 'YYYY-MM') AS month_year,
        SUM(amount) AS monthly_total
    FROM expenses
    WHERE user_id = 1
    GROUP BY TO_CHAR(expense_date, 'YYYY-MM')
) AS monthly_totals;

3. Category Filtering via HAVING Clause

Demonstrates: JOIN, SUM(), GROUP BY, HAVING SUM(amount) > 5000, ORDER BY

SQL HAVING 
SELECT 
    c.name AS category_name,
    c.budget_limit,
    SUM(e.amount) AS total_spent,
    COUNT(e.id) AS transaction_count
FROM expenses e
JOIN categories c ON e.category_id = c.id
WHERE e.user_id = 1
GROUP BY c.id, c.name, c.budget_limit
HAVING SUM(e.amount) > 5000.00
ORDER BY total_spent DESC;

4. Date Function Aggregations (Weekly & Daily)

Demonstrates: TO_CHAR(date, 'IYYY-"W"IW'), MIN(), MAX(), ORDER BY

Date Functions 
SELECT 
    TO_CHAR(expense_date, 'IYYY-"W"IW') AS year_week,
    MIN(expense_date) AS week_start,
    MAX(expense_date) AS week_end,
    SUM(amount) AS weekly_total,
    AVG(amount) AS weekly_average,
    COUNT(id) AS transaction_count
FROM expenses
WHERE user_id = 1
GROUP BY TO_CHAR(expense_date, 'IYYY-"W"IW')
ORDER BY year_week DESC;
